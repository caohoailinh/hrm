Code.gs (Hỗ trợ 9 Sheets + Hồ sơ 360° + Thuế 2026) 
/**
 * GOOGLE APPS SCRIPT BACKEND - HRM & PAYROLL PRO VN 2026
 * Tự động khởi tạo và đồng bộ 9 sheets:
 * Employees, Departments, Positions, Allowances, Deductions, TaxBrackets, Settings, Attendance, Payroll
 */

const SCHEMAS = {
  Employees: [
    'id', 'name', 'gender', 'dob', 'idCard', 'idCardDate', 'idCardPlace',
    'phone', 'email', 'permanentAddr', 'currentAddr', 'education', 'emergencyContact',
    'department', 'position', 'status', 'contractType', 'contractNo', 'joinDate',
    'contractSignDate', 'contractEndDate', 'manager',
    'baseSalary', 'insSalary', 'taxCode', 'socialInsNo', 'healthInsNo',
    'dependents', 'dependentsList', 'bankName', 'bankAccount', 'bankBranch',
    'allowanceIds', 'deductionIds'
  ],
  Departments: ['id', 'name', 'manager', 'description', 'status'],
  Positions: ['id', 'name', 'defaultAllowance', 'note'],
  Allowances: ['id', 'name', 'calcType', 'amount', 'taxable', 'taxFreeCap', 'insurable'],
  Deductions: ['id', 'name', 'type', 'amount', 'taxDeductible'],
  TaxBrackets: ['level', 'from', 'to', 'rate', 'quickSub'],
  Settings: ['key', 'value'],
  Attendance: ['id', 'period', 'empId', 'standardDays', 'actualDays', 'paidLeave', 'unpaidLeave', 'otNormal', 'otWeekend', 'otHoliday', 'note'],
  Payroll: ['id', 'period', 'empId', 'empName', 'department', 'baseSalary', 'insSalary', 'standardDays', 'payableDays', 'daySalary', 'otSalary', 'taxFreeAllowance', 'taxableAllowance', 'grossSalary', 'insurance', 'totalDeduction', 'taxableIncome', 'pitTax', 'bonus', 'otherDeduction', 'netSalary', 'disbStatus', 'paidAt', 'transferNote']
};

const JSON_COLS = ['allowanceIds', 'deductionIds', 'dependentsList'];

function ensureSheets() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  Object.keys(SCHEMAS).forEach(name => {
    let sheet = ss.getSheetByName(name);
    if (!sheet) {
      sheet = ss.insertSheet(name);
      sheet.appendRow(SCHEMAS[name]);
      sheet.getRange(1, 1, 1, SCHEMAS[name].length).setFontWeight('bold').setBackground('#e2e8f0');
    }
  });
}

function readTable(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(name);
  if (!sheet) { ensureSheets(); sheet = ss.getSheetByName(name); }
  const rows = sheet.getDataRange().getValues();
  if (rows.length <= 1) return [];
  const headers = rows[0];
  return rows.slice(1).map(r => {
    let obj = {};
    headers.forEach((h, i) => {
      let val = r[i];
      if (JSON_COLS.indexOf(h) !== -1 && typeof val === 'string') {
        try { val = JSON.parse(val); } catch(e) { val = []; }
      }
      obj[h] = val;
    });
    return obj;
  });
}

function writeTable(name, records) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(name);
  if (!sheet) { ensureSheets(); sheet = ss.getSheetByName(name); }
  const headers = SCHEMAS[name];
  sheet.clearContents();
  sheet.appendRow(headers);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#e2e8f0');
  if (records && records.length > 0) {
    const matrix = records.map(rec => headers.map(h => {
      let v = rec[h];
      if (Array.isArray(v) || (typeof v === 'object' && v !== null)) return JSON.stringify(v);
      return v !== undefined && v !== null ? v : '';
    }));
    sheet.getRange(2, 1, matrix.length, headers.length).setValues(matrix);
  }
}

function doGet(e) {
  ensureSheets();
  const data = {};
  Object.keys(SCHEMAS).forEach(k => { data[k] = readTable(k); });
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', data })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    ensureSheets();
    const body = JSON.parse(e.postData.contents);
    if (body.action === 'SYNC_ALL' && body.data) {
      Object.keys(SCHEMAS).forEach(k => {
        if (body.data[k]) writeTable(k, body.data[k]);
      });
    } else if (body.action === 'SAVE_TABLE' && body.table && body.records) {
      writeTable(body.table, body.records);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
